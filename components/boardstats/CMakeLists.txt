idf_component_register(SRCS "boardstats.c" INCLUDE_DIRS "."
    PRIV_INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}"
    PRIV_REQUIRES esp_wifi esp_netif esp_event esp_http_client esp-tls esp_timer esp_driver_tsens nvs_flash esp_app_format mbedtls spi_flash esp_hw_support)

# The WiFi networks and the dashboard's address and token, from the gitignored
# secrets/ into a header in the BUILD directory (this repo is public).
# Networks: WIFI_SSID/WIFI_PASS, WIFI2_SSID/WIFI2_PASS ... in any of these
# files; the boards that travel get home's and work's alike.
set(_lines "")
foreach(_f stan-claw.env panel1.env tiny2.env boards.env)
    set(_p "${CMAKE_CURRENT_LIST_DIR}/../../secrets/${_f}")
    if(EXISTS "${_p}")
        file(STRINGS "${_p}" _l)
        list(APPEND _lines ${_l})
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_p}")
    endif()
endforeach()
function(bs_escape var)
    string(REPLACE "\\" "\\\\" _v "${${var}}")
    string(REPLACE "\"" "\\\"" _v "${_v}")
    set(${var} "${_v}" PARENT_SCOPE)
endfunction()
set(BS_NETS "")
set(BS_SEEN "")
foreach(_n "" 2 3 4 5 6)
    set(_s "")
    set(_pw "")
    foreach(_l IN LISTS _lines)
        if(_l MATCHES "^WIFI${_n}_SSID=(.*)$")
            set(_s "${CMAKE_MATCH_1}")
        elseif(_l MATCHES "^WIFI${_n}_PASS=(.*)$")
            set(_pw "${CMAKE_MATCH_1}")
        endif()
    endforeach()
    if(NOT _s STREQUAL "" AND NOT "${_s}" IN_LIST BS_SEEN)
        list(APPEND BS_SEEN "${_s}")
        bs_escape(_s)
        bs_escape(_pw)
        string(APPEND BS_NETS "{\"${_s}\", \"${_pw}\"}, ")
    endif()
endforeach()
set(BS_URL "")
set(BS_TOKEN "")
foreach(_l IN LISTS _lines)
    if(_l MATCHES "^BOARDS_URL=(.*)$")
        set(BS_URL "${CMAKE_MATCH_1}")
    elseif(_l MATCHES "^BOARDS_TOKEN=(.*)$")
        set(BS_TOKEN "${CMAKE_MATCH_1}")
    endif()
endforeach()
bs_escape(BS_URL)
bs_escape(BS_TOKEN)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/boardstats_secrets.h.tmp"
     "/* Generated from secrets/ -- never commit. */\n"
     "#define BS_NETS { ${BS_NETS} }\n#define BS_URL \"${BS_URL}\"\n#define BS_TOKEN \"${BS_TOKEN}\"\n")
configure_file("${CMAKE_CURRENT_BINARY_DIR}/boardstats_secrets.h.tmp" "${CMAKE_CURRENT_BINARY_DIR}/boardstats_secrets.h" COPYONLY)
